let rec map f xs =
  match xs with
  | [] -> []
  | y :: ys -> f y :: map f ys

Tw.
map f (map g xs) = map (fun x -> f (g x)) xs.

    1. baza indukcji: weźmy listę pustą i pokażmy że Tw. zachodzi dla []
        L = map f (map g []) = map f [] = [] 
        P = map (fun x -> f (g x)) [] = dowolna fukncja dla [] -> []  
        L = P
    2. krok indukcyjny: weźmy dowolną listę niepustą xs i element x i pokażmy, że Tw. zachodzi dla xs' = x :: xs
    - czyli mamy, że zachodzi: map f (map g xs) = map (fun x -> f (g x)) xs
        L = map f (map g x::xs) = map f (g x :: map g xs) = f (g x) :: map f (map g xs) = f (g x) :: map (fun x -> f (g x)) xs
        P = map (fun x -> f (g x)) x::xs = (fun x -> f (g x)) x :: map (fun f (g x)) xs = f (g x) :: map (fun x -> f (g x)) xs
        L = P 
    Więc Tw. zachodzi na mocy indukcji strukturalnej

type 'v nnf =
    | NNFLit of bool * 'v
    | NNFConj of 'v nnf * 'v nnf
    | NNFDisj of 'v nnf * 'v nnf

Zasada indukcji dla tego typu danych:

Dla każdej własności P wartości typu 'v nnf 
    - jeśli dla dowolnej wartości b typu bool i zmiennej v typu 'v zachodzi P(NNFLit(b, v)) oraz
    - jeśli dla dowolnych dwóch podformuł typu 'v nnf: p' i p'' 
        - z zał. P(p') oraz P(p'') ==> zachodzi P(NNFConj(p', p'')) oraz
        - z zał. P(p') oraz P(p'') ==> zachodzi P(NNFDisj(p', p''))
    to dla każdej formuły p zachodzi P.

type 'v formula =
| Var of 'v
| Neg of 'v formula
| Conj of 'v formula * 'v formula
| Disj of 'v formula * 'v formula


let to_nnf p = 
  let rec it p flaga = 
    match p with 
    | Var v -> NNFLit (flaga, v)
    | Neg v -> it p (not flaga)
    | Conj (p', p'') -> 
        if flaga = true then NNFDisj (it p' true, it p'' true)
        else NNFConj (it p' false, it p'' false)
    | Disj (p', p'') ->
        if flaga = true then NNFConj (it p' true, it p'' true)
        else NNFDisj (it p' false, it p'' false)
    in it p false

  match xs with
  | [] -> true
  | [x] -> true
  | x :: y :: ys -> x <= y && is_sorted (y :: ys)
