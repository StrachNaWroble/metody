ZADANIE 3.

module MMap = Map.Make(String)

module Env : sig
  type t

  val empty    : t
  val add : t -> Ast.var -> t
  val push_tmp : t -> t
  val lookup : t -> Ast.var -> int
  
end = struct
  type t = { map : int MMap.t; depth : int; }

  let empty     = { map = MMap.empty; depth = 0 }
  let add env x = { map = MMap.add x env.depth env.map; depth = env.depth + 1 }
  let push_tmp env = { map = env.map; depth = env.depth + 1 } 
  let lookup env x = 
  	match MMap.find_opt x env.map with
  	| Some i -> env.depth - i - 1
  	| _ -> failwith ("Unbound variable " ^ x) 
end
