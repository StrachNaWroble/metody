module DictList : DICT = struct 
type ('a, 'b) dict = ('a * 'b) list
let empty = []
let insert a b d = (a, b) :: d
let remove a d = List.filter (fun (ap, bp) -> ap <> a) d

    match d with 
    | [] -> None
    | (ap, bp) :: rest ->
        if ap = a then Some bp
        else find_opt a rest
        
let rec find a d =
    match find_opt a d with 
    | Some b -> b
    | None -> failwith "Nie ma takiej wartosci"

module TreeDict : DICT = struct
type ('a, 'b) dict = 
    | Leaf 
    | Node of ('a, 'b) dict * 'a * 'b * ('a, 'b) dict
    
let empty = Leaf

let rec insert a b d = 
    match d with
    | Leaf -> Node (Leaf, a, b, Leaf)
    | Node (vl, ap, bp, vr) ->
        if ap = a then Node (vl, ap, b, vr) else
        if ap > a then Node (vl, ap, bp, insert a b vr) else 
        Node (insert a b vl, ap, bp, vr)
        
let rec find_opt a d = 
    match d with 
    | Leaf -> None
    | Node (vl, ap, bp, vr) ->
        if a = ap then Some bp else
        if ap > a then find_opt a vr else
        find_opt a vl 

let find a d = 
    match find_opt a d with
    | None -> failwith "Nie ma takiej wartosci"
    | Some b -> b

let rec append d acc = 
    match d with 
    | Leaf -> acc
    | Node (vl, a, b, vr) -> 
        let r = append vr acc in
        let ab_r = (a, b) :: r in
        append vl ab_r

let to_list d = append d []

let rec min_node d = 
    match d with 
    | Leaf -> failwith "nie może być takiej sytuacji"
    | Node (Leaf, a, b, _) -> (a, b)
    | Node (vl, _, _, _) -> min_node vl

let rec remove a d = 
    match d with
    | Leaf -> Leaf
    | Node (vl, ap, bp, vr) ->
        if ap > a then Node (remove a vl, ap, bp, vr)
        else if ap < a then Node (vl, ap, bp, remove a vr)
        else 
            match vl, vr with
            | Leaf, rest -> rest
            | rest, Leaf -> rest
            | _ -> 
                let (new_a, new_b) = min_node vr in 
                Node (vl, new_a, new_b, remove new_a vr)
end

module Freq (D : DICT) = struct
let freq (xs : 'a list) : ('a * int) list =
    let rec it a d =
    match a with
    | [] -> D.to_list d
    | head :: tail ->
        match (D.find_opt head d) with
        | None -> it tail (D.insert head 1 d)
        | Some b -> it tail (D.insert head (b+1) d)
    in it xs D.empty

let list_of_string s = String.to_seq s |> List.of_seq

let char_freq s = 
	freq (list_of_string s)
		
end
